export const getExp = () => {
  const start = new Date('2018-01-01');
  const now = new Date();
  let diff = now.getFullYear() - start.getFullYear();
  if (now.getMonth() < start.getMonth() || (now.getMonth() === start.getMonth() && now.getDate() < start.getDate())) {
    diff--;
  }
  return diff;
};
